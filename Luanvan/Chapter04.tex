\chapter{MÔ HÌNH VÀ PHƯƠNG PHÁP ĐỀ XUẤT}

\section{Tổng quan kiến trúc hệ thống đề xuất}

Kiến trúc System-on-Chip (SoC) đề xuất được thiết kế nhằm tối ưu hóa băng thông giao tiếp ngoại vi và kiểm chứng khả năng truyền dữ liệu theo cơ chế Double Data Rate (DDR) trên nền tảng FPGA. Trung tâm của hệ thống là lõi xử lý RISC-V PicoRV32 đảm nhiệm vai trò điều khiển và phát các yêu cầu truy cập bộ nhớ. Để đảm bảo tính đóng gói, hiệu năng truyền tải cao và khả năng xử lý song song, hạ tầng bus AMBA AXI4-Lite được chọn làm tuyến bus nội bộ kết nối CPU với các mô-đun ngoại vi thay vì bus APB truyền thống \cite{picorv32_repo,amd_axi4lite_hls,jedec_ddr}.

Nhằm khẳng định rõ đóng góp khoa học của đề tài cũng như đảm bảo tính minh bạch học thuật, phạm vi thiết kế phần cứng được phân định cụ thể thành hai nhóm thành phần chính:
\begin{itemize}
    \item {Các khối kế thừa từ mã nguồn mở:} Lõi RISC-V PicoRV32 đảm nhiệm vai trò CPU trung tâm, còn BRAM chứa mã chương trình khởi tạo (\path{firmware.hex}). Việc kế thừa giúp bảo đảm tính chuẩn hóa của hạ tầng xử lý lệnh và độ tin cậy khi thực thi phần mềm \cite{picorv32_repo,riscv_unprivileged}.
    \item {Các khối tự thiết kế và tích hợp:} Hệ thống gồm bộ chuyển đổi \path{picorv32_axi_adapter}, bộ định tuyến AXI Crossbar đa kênh, khối xử lý song song đa làn và bộ phát dữ liệu hai cạnh clock. Các thành phần được thiết kế ở mức RTL bằng Verilog HDL và đóng gói vào SoC.
\end{itemize}

Toàn bộ hệ thống được chia thành hai miền chức năng chính gồm miền xử lý trung tâm và miền ngoại vi AXI. PicoRV32 lấy lệnh từ BRAM nội bộ thông qua giao diện Native Memory; các giao dịch dữ liệu đến ngoại vi được chuyển sang chuẩn AXI4-Lite bởi khối \path{picorv32_axi_adapter}. Hệ thống ngoại vi gồm UART, SPI, I2C và khối parallel. Khối parallel vừa thực hiện bốn phép cộng độc lập vừa phát bốn kết quả đó dưới dạng luồng DDR 32 bit. \section{Thiết kế chi tiết các khối chức năng}

\subsection{Lõi xử lý PicoRV32 và bộ giải mã địa chỉ}

Lõi xử lý PicoRV32 được cấu hình hoạt động theo tập lệnh RV32I cơ bản, sử dụng giao diện Native Memory để trao đổi dữ liệu với bên ngoài. Giao diện này bao gồm các tín hiệu địa chỉ \path{mem_addr}, dữ liệu ghi \path{mem_wdata}, dữ liệu đọc \path{mem_rdata}, tín hiệu cho phép \path{mem_valid}, tín hiệu phản hồi \path{mem_ready} và tín hiệu chọn byte \path{mem_wstrb} \cite{picorv32_repo,riscv_unprivileged}.

Không gian địa chỉ 32 bit được phân bổ như trong Bảng~\ref{tab:soc-memory-map}. Vùng địa chỉ thấp dành cho BRAM 4 KB, trong đó tệp \path{firmware.hex} được nạp làm chương trình khởi động. Các giao dịch đến vùng ngoại vi từ \path{0x4000_0000} đến \path{0x4000_3FFF} được chuyển qua AXI Adapter.

Tại phía Slave, mô-đun \path{soc_axi_top} đóng vai trò là một bộ định tuyến (Crossbar) giải mã địa chỉ nhằm dẫn luồng 5 kênh giao dịch của AXI đến đúng thiết bị ngoại vi tương ứng.

\subsection{Bộ chuyển đổi bus \texorpdfstring{picorv32\_axi\_adapter}{picorv32 axi adapter}}

Khối \path{picorv32_axi_adapter} chuyển các giao dịch dữ liệu của giao diện Native Memory sang chuẩn AMBA AXI4-Lite. Giao dịch lấy lệnh được nhận biết bằng \path{mem_instr} và trả về trực tiếp từ BRAM qua \path{bram_rdata} trong vòng 1 chu kỳ; giao dịch dữ liệu được biên dịch thành các kênh giao tiếp độc lập (Kênh Ghi AW/W/B và Kênh Đọc AR/R). Bộ chuyển đổi sử dụng máy trạng thái FSM với ba trạng thái cốt lõi: \path{S_IDLE}, \path{S_WRITE_WAIT} và \path{S_READ_WAIT}. Cách tổ chức này bảo đảm việc đàm phán tín hiệu (Handshake) `VALID/READY` trên bus AXI được duy trì ổn định cho đến khi ngoại vi xử lý xong \cite{picorv32_repo,amd_axi4lite_hls}.

\begin{figure}[H]
\centering
\includegraphics[width=0.72\textwidth,height=0.45\textheight,keepaspectratio]{images/AXI_RTL.png}
\caption{Sơ đồ RTL các kênh tín hiệu của bộ chuyển đổi \texttt{picorv32\_axi\_adapter}.}
\label{fig:rtl-axi-adapter}
\end{figure}

Hình~\ref{lst:axi-adapter-fsm} trình bày phần RTL tương ứng của máy trạng thái trong thiết kế.

\begin{lstlisting}[language=Verilog, caption={Máy trạng thái chuyển đổi Native Memory sang AXI4-Lite trong picorv32\_axi\_adapter.}, label={lst:axi-adapter-fsm}]
always @(posedge clk or negedge resetn) begin
    if (!resetn) begin
        state <= S_IDLE; mem_ready <= 1'b0;
        mem_axi_awvalid <= 0; mem_axi_wvalid <= 0; mem_axi_arvalid <= 0;
    end else begin
        mem_ready <= 1'b0;
        case (state)
            S_IDLE: begin
                if (mem_valid && is_mmio) begin
                    if (|mem_wstrb) begin // Giao dich Ghi
                        mem_axi_awvalid <= 1'b1;
                        mem_axi_wvalid  <= 1'b1;
                        state           <= S_WRITE_WAIT;
                    end else begin        // Giao dich Doc
                        mem_axi_arvalid <= 1'b1;
                        state           <= S_READ_WAIT;
                    end
                end
            end
            S_WRITE_WAIT: begin
                // Xoa co VALID khi Slave bao READY
                if (mem_axi_awready) mem_axi_awvalid <= 1'b0;
                if (mem_axi_wready)  mem_axi_wvalid  <= 1'b0;
                // Cho Slave tra ve BVALID (Hoan tat ghi)
                if (mem_axi_bvalid) begin
                    mem_ready <= 1'b1;
                    state     <= S_IDLE;
                end
            end
            S_READ_WAIT: begin
                if (mem_axi_arready) mem_axi_arvalid <= 1'b0;
                // Cho Slave tra ve RVALID (Du lieu doc hop le)
                if (mem_axi_rvalid) begin
                    mem_rdata <= mem_axi_rdata;
                    mem_ready <= 1'b1;
                    state     <= S_IDLE;
                end
            end
        endcase
    end
end
\end{lstlisting}

\subsubsection*{Cơ chế hoạt động và phân tích chi tiết mã nguồn FSM}

Đoạn mã trong Hình~\ref{lst:axi-adapter-fsm} thực hiện quá trình chuyển đổi giao thức (Protocol Translation) giữa giao diện bus Native Memory của PicoRV32 và bus AMBA AXI4-Lite. Luồng xử lý chi tiết diễn ra như sau:

\begin{enumerate}
    \item \textbf{Trạng thái rảnh (\path{S_IDLE}):} Mạch giám sát tín hiệu \path{mem_valid}. Khi phát hiện truy xuất đến vùng ngoại vi (\path{is_mmio}), nếu đây là giao dịch Ghi (\path{|mem_wstrb > 0}), FSM kích hoạt song song kênh địa chỉ (\path{AWVALID}) và kênh dữ liệu (\path{WVALID}) rồi chuyển sang \path{S_WRITE_WAIT}. Nếu là giao dịch Đọc, FSM kích hoạt kênh đọc địa chỉ (\path{ARVALID}) và rẽ nhánh sang \path{S_READ_WAIT}.
    \item \textbf{Trạng thái chờ Ghi (\path{S_WRITE_WAIT}):} Adapter theo dõi cờ \path{READY} của Slave trên cả hai kênh AW và W. Ngay khi Slave chấp nhận, cờ \path{VALID} tương ứng sẽ được hạ xuống. Giao dịch chỉ hoàn tất và trả tín hiệu \path{mem_ready = 1} về CPU khi Slave phản hồi thành công qua kênh B (\path{BVALID = 1}).
    \item \textbf{Trạng thái chờ Đọc (\path{S_READ_WAIT}):} Tương tự, Adapter chờ Slave phản hồi dữ liệu qua kênh R. Khi \path{RVALID = 1}, dữ liệu \path{mem_axi_rdata} được chép vào \path{mem_rdata} và kết thúc giao dịch.
\end{enumerate}

Tác dụng chính của khối Adapter là bóc tách một giao dịch đơn khối của CPU thành luồng tín hiệu 5 kênh độc lập của AXI4-Lite, giải quyết bài toán đồng bộ hóa và chống hiện tượng vấp tín hiệu (glitch) trong hệ thống.

\subsection{Khối xử lý song song và truyền dữ liệu DDR}

Khối parallel được ánh xạ tại vùng \path{0x4000_3000--0x4000_3FFF} và đóng vai trò bộ tăng tốc song song đa làn. Khối tích hợp đầu ra DDR để thực hiện đồng thời các phép tính do CPU giao và tuần tự hóa kết quả phát ra ngoài chip.

\subsubsection*{1. Cơ chế ghi/đọc thanh ghi và kích hoạt tính toán}
Hệ thống bố trí tập thanh ghi AXI để CPU tương tác với khối xử lý song song:
\begin{itemize}
    \item {Thanh ghi toán hạng A (Offset \path{0x10--0x1C}):} Chứa 4 giá trị toán hạng $A[0]$ đến $A[3]$ (mỗi giá trị 32 bit).
    \item {Thanh ghi toán hạng B (Offset \path{0x20--0x2C}):} Chứa 4 giá trị toán hạng $B[0]$ đến $B[3]$ (mỗi giá trị 32 bit).
    \item {Thanh ghi điều khiển START (Offset \path{0x00}):} Khi CPU ghi bit \path{1} vào thanh ghi này, tín hiệu kích hoạt được phát ra để 4 lane đồng thời thực hiện phép cộng 32 bit: $Result[i] = A[i] + B[i]$ (với $i = 0 \dots 3$).
    \item {Thanh ghi kết quả (Offset \path{0x30--0x3C}):} Chứa 4 kết quả tính toán. Đường đọc kênh R của AXI cho phép CPU truy xuất lại các thanh ghi này bất kỳ lúc nào.
\end{itemize}

\subsubsection*{2. Cơ chế đóng gói và phân tích mã nguồn bộ phát DDR}

Song song với việc lưu kết quả vào thanh ghi để CPU đọc, toàn bộ 128 bit kết quả (4 làn $\times$ 32 bit) được nạp vào bộ tuần tự hóa \path{ddr_result_serializer} nhằm đẩy ra các chân I/O ngoài chip qua 2 chu kỳ clock (tương ứng 4 cạnh xung nhịp).

Hình~\ref{lst:parallel-ddr-serializer} thể hiện đoạn mã kết nối giữa luồng kết quả tính toán và khối tuần tự hóa DDR.

\begin{lstlisting}[language=Verilog, caption={Giao tiếp giữa kết quả các làn tính toán và bộ phát DDR.}, label={lst:parallel-ddr-serializer}]
// Generate the load condition for the DDR serializer
assign fire_start = start_pending & core_in_ready & {NUM_LANES{ddr_ready}};

// Serialize four 32-bit results over four clock edges
ddr_result_serializer #(.DATA_WIDTH(32)) u_ddr_result_serializer (
    .clk         (clk),
    .rst_n       (rst_n),
    .load        (|fire_start),
    .data_rise_0 (result_flat[31:0]),
    .data_fall_0 (result_flat[63:32]),
    .data_rise_1 (result_flat[95:64]),
    .data_fall_1 (result_flat[127:96]),
    .ddr_data    (ddr_data),
    .ddr_valid   (ddr_valid),
    .ddr_done    (ddr_done),
    .ddr_ready   (ddr_ready)
);
\end{lstlisting}

{Giải thích đoạn mã trong Hình~\ref{lst:parallel-ddr-serializer}:}
\begin{enumerate}
    \item {Dòng 2 (\path{assign fire_start}):} Tín hiệu \path{fire_start} đóng vai trò là điều kiện bắt tay (handshake). Lệnh tính toán chỉ thực sự được nạp khi có yêu cầu thực thi (\path{start_pending}), lõi tính toán rảnh (\path{core_in_ready}), và quan trọng nhất là khối phát DDR đang sẵn sàng (\path{ddr_ready = 1}). Điều này ngăn chặn hiện tượng ghi đè dữ liệu khi batch trước chưa phát xong.
    \item {Dòng 8 (\path{.load}):} Sử dụng phép OR thu gọn (\path{|fire_start}) để tạo xung nạp dữ liệu. Khi có hiệu lệnh, 128 bit kết quả ghép từ 4 lane (\path{result_flat}) lập tức được chốt vào các thanh ghi đầu vào của bộ phát.
    \item {Dòng 9--12 (\path{.data_rise_X}, \path{.data_fall_X}):} Mảng dữ liệu 128 bit được chia thành 4 từ dữ liệu 32 bit:
    \begin{itemize}
        \item \path{data_rise_0} (Lane 0): Phát tại {cạnh lên} của chu kỳ clock thứ nhất.
        \item \path{data_fall_0} (Lane 1): Phát tại {cạnh xuống} của chu kỳ clock thứ nhất.
        \item \path{data_rise_1} (Lane 2): Phát tại {cạnh lên} của chu kỳ clock thứ hai.
        \item \path{data_fall_1} (Lane 3): Phát tại {cạnh xuống} của chu kỳ clock thứ hai.
    \end{itemize}
    \item {Dòng 13--16 (\path{ddr_data}, \path{ddr_valid}, \path{ddr_done}, \path{ddr_ready}):} Quản lý luồng đầu ra. \path{ddr_valid = 1} kéo lên trong suốt 2 chu kỳ phát dữ liệu; \path{ddr_done} phát ra một xung ngắn đánh dấu hoàn tất truyền dữ liệu của một đợt tính toán.
\end{enumerate}

\subsubsection*{3. Nguyên lý phát dữ liệu hai cạnh xung nhịp qua Primitive ODDR}

Để đưa bus dữ liệu 32 bit ra chân vật lý của FPGA với tốc độ gấp đôi mà không cần tăng gấp đôi tần số clock hệ thống, thiết kế sử dụng phần tử phần cứng chuyên dụng Output Double Data Rate (\path{ODDR}) nằm tại khối Input/Output Block (IOB) của chip AMD/Xilinx \cite{amd_ug471}.

Hình~\ref{lst:oddr-output} mô tả cách khởi tạo primitive \path{ODDR} cho từng đường truyền dữ liệu.

\begin{lstlisting}[language=Verilog, caption={Khởi tạo phần tử cứng ODDR trong AMD/Xilinx FPGA.}, label={lst:oddr-output}]
// Instantiate one ODDR primitive for each output bit
ODDR #(
    .DDR_CLK_EDGE ("SAME_EDGE"),
    .INIT         (1'b0),
    .SRTYPE       ("ASYNC")
) u_oddr (
    .Q  (ddr_data[bit_index]),
    .C  (clk),
    .CE (1'b1),
    .D1 (rise_word[bit_index]),
    .D2 (fall_word[bit_index]),
    .R  (~rst_n),
    .S  (1'b0)
);
\end{lstlisting}

\textbf{Phân tích chuyên sâu về phần tử cứng ODDR (Hình~\ref{lst:oddr-output}):}
\begin{itemize}
    \item \textbf{Cấu hình \path{DDR_CLK_EDGE = "SAME_EDGE"}:} Trong chế độ này, dữ liệu đưa vào \path{D1} và \path{D2} được tổ chức để phát lần lượt tại cạnh lên và cạnh xuống của xung nhịp \path{C}, trong khi việc nạp dữ liệu được đồng bộ theo cùng một cạnh clock. Cấu hình này đơn giản hóa đường định thời dữ liệu trong lớp logic FPGA \cite{amd_ug471}.
    \item \textbf{Ngõ ra \path{Q}:} Được nối thẳng tới chân I/O vật lý của chip FPGA (\path{ddr_data[bit_index]}). Nhờ sử dụng primitive IOB chuyên dụng thay vì dùng Flip-Flop thông thường trong lát logic (CLB), độ trễ đường truyền từ thanh ghi ra chân chip được giảm thiểu và có tính đồng đều cao giữa các đường truyền của bus 32 bit.
\end{itemize}

\subsection{Các khối ngoại vi chuẩn giao tiếp nối tiếp (UART, SPI, I2C)}

Bên cạnh khối xử lý song song và đầu ra DDR, hệ thống SoC còn tích hợp các khối ngoại vi chuẩn giao tiếp nối tiếp đóng vai trò là giao diện kết nối giữa SoC với các linh kiện ngoài chip và hệ thống giám sát:

\begin{itemize}
    \item \text{Khối AXI UART Slave (\path{0x4000_0000}):} Đảm nhiệm vai trò truyền nhận dữ liệu chuỗi bất đồng bộ với máy tính chủ (Host PC) phục vụ mục đích truyền nhận dữ liệu kiểm thử và xuất log điều khiển (debug).
    \item \text{Khối AXI SPI Slave (\path{0x4000_1000}):} Cung cấp giao diện truyền dữ liệu nối tiếp đồng bộ tốc độ cao, hỗ trợ kết nối với các thiết bị ngoại vi như bộ nhớ Flash ngoài hoặc màn hình hiển thị.
    \item \text{Khối AXI I2C Slave (\path{0x4000_2000}):} Cung cấp giao diện giao tiếp 2 dây chuẩn (SDA, SCL) phục vụ việc đọc/ghi dữ liệu cấu hình từ các cảm biến hoặc IC quản lý nguồn.
\end{itemize}

Tất cả các mô-đun ngoại vi trên đều được đóng gói chuẩn giao diện AXI4-Lite, kết nối trực tiếp vào bộ định tuyến Crossbar của \path{soc_axi_top} và được quản lý thông qua tập thanh ghi cấu hình tương ứng trong bảng bản đồ bộ nhớ (Bảng~\ref{tab:soc-memory-map}).

\subsubsection{Khối giao tiếp nối tiếp bất đồng bộ (AXI UART Slave)}

\begin{figure}[H]
\centering
\IfFileExists{images/UART_RTL.png}{%
    \makebox[\textwidth][c]{\includegraphics[width=1.08\textwidth]{images/UART_RTL.png}}%
}{%
    \fbox{\parbox[c][5cm][c]{0.85\textwidth}{\centering Không tìm thấy tệp images/UART\_RTL.png}}%
}
\caption{Sơ đồ RTL của khối giao tiếp AXI UART.}
\label{fig:rtl-uart}
\end{figure}

Mô-đun \path{axi_uart_slave} thực hiện việc truyền nhận dữ liệu bất đồng bộ chuẩn UART. Kiến trúc chính bao gồm giao diện AXI4-Lite, bộ phát (\path{TX}) và bộ nhận (\path{RX}) độc lập. Cấu trúc khung truyền và nguyên tắc xác định tốc độ baud được xây dựng theo đặc tả UART thông dụng \cite{ti_uart_overview}. Các đoạn mã Verilog cốt lõi được thể hiện trong Hình~\ref{lst:axi-uart-slave}.

\begin{lstlisting}[language=Verilog, caption={Các đoạn mã RTL cốt lõi của mô-đun axi\_uart\_slave.}, label={lst:axi-uart-slave}]
// 1. Gia ma giao dien AXI Write Handshake
wire slv_reg_wren = axi_wready && S_AXI_WVALID && axi_awready && S_AXI_AWVALID;
always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) begin
        tx_data <= 8'h0; tx_start <= 1'b0;
    end else begin
        tx_start <= 1'b0;
        if (slv_reg_wren && (S_AXI_AWADDR[7:0] == 8'h00)) begin
            tx_data  <= S_AXI_WDATA[7:0];
            tx_start <= 1'b1; // Kich hoat truyen TX
        end
    end
end

// 2. TX shift register (PISO)
always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) begin ... end
    else if (tx_start && !tx_active) begin
        tx_active    <= 1'b1;
        tx_shift_reg <= {1'b1, tx_data, 1'b0}; // Khung: Stop(1) + Data(8) + Start(0)
    end else if (tx_active && clk_cnt == BIT_PERIOD - 1) begin
        tx_shift_reg <= tx_shift_reg >> 1;     // Dich phai tung bit
        if (bit_idx == 9) tx_active <= 1'b0;
    end
end

// 3. Two-stage RX synchronizer and start-bit detector
reg [1:0] rx_sync;
always @(posedge S_AXI_ACLK) rx_sync <= {rx_sync[0], uart_rx};

always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) begin ... end
    else begin
        if (rx_clear) rx_done <= 1'b0;
        if (!rx_active && (rx_sync[1] == 1'b0)) begin // Phat hien canh xuong Start bit
            rx_active <= 1'b1;
        end else if (rx_active && rx_clk_cnt == BIT_PERIOD - 1) begin
            if (rx_bit_idx >= 1 && rx_bit_idx <= 8)
                rx_shift_data <= {rx_sync[1], rx_shift_data[7:1]}; // Lay mau du lieu RX
        end
    end
end
\end{lstlisting}

Giải thích mã nguồn khối \path{axi_uart_slave}:
\begin{itemize}
    \item \text{Giải mã AXI Write:} Tín hiệu ghi dữ liệu \path{slv_reg_wren} chỉ được bật lên khi kênh AW và kênh W cùng thiết lập trạng thái bắt tay thành công (\path{VALID} và \path{READY} đồng thời bằng 1).
    \item \text{Phân giải baud rate:} Tốc độ truyền được tính toán qua hằng số \path{BIT_PERIOD = CLK_FREQ / BAUD_RATE}. Mạch đếm \path{clk_cnt} chia tần số để định thời chính xác khoảng thời gian của 1 bit.
    \item \text{Khối truyền (\path{TX}):} Thanh ghi dịch \path{tx_shift_reg} 10 bit được nạp định dạng bao gồm 1 bit Start (0), 8 bit dữ liệu \path{tx_data}, và 1 bit Stop (1). Dữ liệu được dịch phải tuần tự ra đường phát \path{uart_tx}.
    \item \text{Khối nhận (\path{RX}) và chống hiện tượng bất ổn định:} Đường vào \path{uart_rx} được đưa qua thanh ghi đồng bộ 2 tầng \path{rx_sync} nhằm loại bỏ hiện tượng bất ổn định trạng thái (Metastability) khi tín hiệu bất đồng bộ đi vào miền xung nhịp \path{S_AXI_ACLK}. Khi phát hiện cạnh xuống (bit Start), mạch nhận thực hiện dịch 8 bit dữ liệu vào \path{rx_shift_data} và kích hoạt cờ \path{rx_done}. Cờ này tự động xóa (\path{rx_clear}) khi CPU thực hiện thao tác đọc thanh ghi dữ liệu nhận qua kênh R của bus AXI.
\end{itemize}

    \fbox{\parbox[c][5cm][c]{0.85\textwidth}{\centering Không tìm thấy tệp images/SPI\_RTL.png}}%

\end{figure}

Mô-đun \path{axi_spi_slave} chuyển đổi các truy xuất AXI4-Lite thành giao thức truyền nối tiếp 4 dây SPI Master. Quan hệ giữa cực tính, pha xung nhịp và thời điểm lấy mẫu dữ liệu được thiết lập theo SPI Mode 0 \cite{analog_spi_overview}. Các đoạn mã FSM và dịch dữ liệu quan trọng được trình bày tại Hình~\ref{lst:axi-spi-slave}.

\begin{lstlisting}[language=Verilog, caption={Các đoạn mã RTL cốt lõi của mô-đun axi\_spi\_slave.}, label={lst:axi-spi-slave}]
// 1. Gia ma thanh ghi Chip Select (CS) qua AXI
wire slv_reg_wren = axi_wready && S_AXI_WVALID && axi_awready && S_AXI_AWVALID;
always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) spi_cs <= 1'b1;
    else if (slv_reg_wren && S_AXI_AWADDR[7:0] == 8'h04)
        spi_cs <= S_AXI_WDATA[0]; // Dieu khien spi_cs bang phan mem
end

// 2. May trang thai (FSM) dieu khien SCLK va truyen nhan SPI (Mode 0)
always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) begin ... end
    else begin
        case (state)
            0: begin // IDLE
                spi_sclk <= 1'b0;
                if (start_tx && !spi_cs) begin
                    spi_busy   <= 1'b1;
                    tx_shifter <= tx_data_buf;
                    state      <= 1;
                end
            end
            1: begin // Canh len SCLK - Lay mau du lieu MISO
                spi_sclk <= 1'b1;
                if (clk_div == 8'd5) begin
                    rx_data_buf <= {rx_data_buf[6:0], spi_miso};
                    state       <= 2;
                end
            end
            2: begin // Falling SCLK edge: drive the next MOSI bit
                spi_sclk <= 1'b0;
                if (clk_div == 8'd5) begin
                    if (bit_cnt < 7) begin
                        bit_cnt    <= bit_cnt + 1;
                        tx_shifter <= tx_shifter << 1; // Dich bit MSB ra MOSI
                        state      <= 1;
                    end else state  <= 0; // Hoan thanh 8-bit
                end
            end
        endcase
    end
end
\end{lstlisting}

Giải thích mã nguồn khối \path{axi_spi_slave}:
\begin{itemize}
    \item {Điều khiển chọn chip:} Phần mềm điều khiển \path{spi_cs} bằng thanh ghi tại offset \path{0x04} qua tín hiệu \path{S_AXI_WDATA}. Khi \path{spi_cs} bằng 0, giao tiếp SPI sẵn sàng hoạt động.
    \item {Máy trạng thái SPI:} FSM dùng bộ đếm \path{clk_div} để tạo tín hiệu \path{spi_sclk}.
    \item {Truyền nhận Mode 0:} Ở cạnh lên của \path{spi_sclk}, dữ liệu \path{spi_miso} được chốt vào \path{rx_data_buf}. Ở cạnh xuống, \path{tx_shifter} đẩy bit kế tiếp ra \path{spi_mosi}. Sau 8 bit, FSM trở về trạng thái nghỉ và hạ cờ \path{spi_busy}.
\end{itemize}

\centering

    \makebox[\textwidth][c]{\includegraphics[width=1.08\textwidth]{images/I2C_RTL.png}}%
}{%
    \fbox{\parbox[c][5cm][c]{0.85\textwidth}{\centering Không tìm thấy tệp images/I2C\_RTL.png}}%
}
\caption{Sơ đồ RTL của khối giao tiếp AXI I\textsuperscript{2}C.}
\label{fig:rtl-i2c}
\end{figure}

Mô-đun \path{axi_i2c_slave} tạo giao diện giao tiếp I2C Master. Các điều kiện START, STOP, chu kỳ ACK và đặc tính ngõ ra kiểu open-drain được triển khai theo đặc tả I2C-bus \cite{nxp_um10204}. Đoạn mã trọng tâm điều khiển các trạng thái và chân ba trạng thái được thể hiện trong Hình~\ref{lst:axi-i2c-slave}.

\begin{lstlisting}[language=Verilog, caption={Các đoạn mã RTL cốt lõi của mô-đun axi\_i2c\_slave.}, label={lst:axi-i2c-slave}]
// 1. May trang thai dieu khien giao thuc I2C (START, STOP, DATA, ACK)
always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
    if (!S_AXI_ARESETN) begin ... end
    else begin
        case (state)
            0: begin // IDLE: Tra SDA va SCL ve muc cao
                scl_out <= 1'b1; sda_out <= 1'b1; sda_oe <= 1'b1;
                if (cmd_trigger) begin
                    if (i2c_cmd[0])      state <= 1; // Lenh START
                    else if (i2c_cmd[1]) state <= 3; // Lenh STOP
                    else if (i2c_cmd[2]) begin bit_idx <= 7; state <= 5; end // Ghi Byte
                end
            end

            // Tao dieu kien START (SDA gia xuong khi SCL dang cao)
            1: begin sda_out <= 1'b0; if (clk_cnt == 50) state <= 2; end
            2: begin scl_out <= 1'b0; if (clk_cnt == 50) state <= 0; end

            // Truyen 8-bit du lieu (MSB truoc)
            5: begin
                sda_out <= i2c_data[bit_idx];
                if (clk_cnt == 25) scl_out <= 1'b1;
                if (clk_cnt == 50) begin
                    scl_out <= 1'b0;
                    if (bit_idx > 0) bit_idx <= bit_idx - 1;
                    else state <= 6; // Chuyen sang kiem tra ACK
                end
            end
            
            // Release SDA and sample the ACK bit from the slave
            6: begin
                sda_oe <= 1'b0; // FPGA tha noi chan SDA (Tri-state)
                if (clk_cnt == 25) begin scl_out <= 1'b1; ack_reg <= sda_in; end // Lay mau ACK
                if (clk_cnt == 50) begin scl_out <= 1'b0; state <= 0; end
            end
        endcase
    end
end
\end{lstlisting}

Giải thích nguồn khối \path{axi_i2c_slave}:
\begin{itemize}
    \item {Điều khiển SDA ba trạng thái:} Tín hiệu \path{sda_oe} điều khiển tầng đệm ở wrapper trên cùng. Khi tín hiệu bằng 1, FPGA chủ động điều khiển SDA; khi bằng 0, FPGA nhả đường SDA để thiết bị slave phản hồi ACK.
    \item {Thực thi lệnh bằng FSM:} Máy trạng thái giải mã thanh ghi \path{i2c_cmd} được nạp từ bus AXI:
    \begin{itemize}
        \item {START (trạng thái 1--2):} Tạo cạnh xuống trên SDA khi SCL đang ở mức cao.
        \item {STOP (trạng thái 3--4):} Tạo cạnh lên trên SDA khi SCL đang ở mức cao.
        \item {Truyền byte và đọc ACK (trạng thái 5--6):} Dịch dữ liệu từ MSB đến LSB, sau đó lấy mẫu \path{sda_in} vào \path{ack_reg} tại cạnh lên của SCL.
    \end{itemize}
\end{itemize}

\section{Phương pháp xử lý Timing và ràng buộc XDC trên Vivado}

Việc phát dữ liệu tại cả hai cạnh của xung nhịp làm khoảng thời gian giữa hai lần thay đổi dữ liệu bị thu hẹp xuống xấp xỉ $T_{clk}/2$ (tương ứng $5.0~\text{ns}$ tại tần số $100~\text{MHz}$). Do đó, phân tích định thời phải bao phủ cả yêu cầu thiết lập và giữ của các đường tới ô \path{ODDR} cũng như giao tiếp từ IOB đến thiết bị nhận \cite{amd_ug471,amd_ug903}.

Phân tích mã nguồn RTL tập trung vào khối parallel và bộ tuần tự hóa DDR cho thấy thiết kế hiện tại bộc lộ ba điểm hạn chế về định thời phần cứng cần được nâng cấp khi triển khai thực tế:
\begin{enumerate}
    \item \textbf{Đường tới hạn (Critical Path) chưa có tầng chốt Pipeline:} Kết quả từ các bộ cộng tổ hợp $A[i] + B[i]$ trong parallel\_lane được kết nối trực tiếp vào các cổng vào \path{data_rise_X} và \path{data_fall_X} của khối phát DDR trong cùng chu kỳ kích hoạt \path{fire_start}. Đường mạch tổ hợp kéo dài từ thanh ghi toán hạng qua bộ cộng 32-bit đến ô \path{ODDR} tạo ra trễ mạch tổ hợp ($t_{logic}$) lớn, dễ gây vi phạm thời gian thiết lập ($t_{setup}$) khi hệ thống vận hành ở tần số cao.
    \item \textbf{Thiếu tín hiệu xung nhịp đồng hành (Forwarded Clock Out):} Dữ liệu DDR phát ra chân chip nhưng chưa có đường xung nhịp đồng bộ đi kèm. Do ảnh hưởng của trễ đệm I/O (IOB delay) biến thiên theo nhiệt độ và điện áp, phía thiết bị nhận bên ngoài sẽ khó xác định chính xác vị trí cạnh xuống để lấy mẫu nửa chu kỳ sau nếu không có một đường clock đồng dạng truyền kèm.
    \item \textbf{Rủi ro cạnh tranh dữ liệu (Race Condition) trong FSM:} Trong bộ tuần tự hóa DDR, tại trạng thái \path{PHASE_FIRST}, từ dữ liệu \path{rise_word} và \path{fall_word} bị ghi đè ngay ở cạnh lên xung nhịp trong khi \path{ODDR} đang lấy mẫu cặp dữ liệu đầu tiên. Việc cập nhật dữ liệu ngay tại thời điểm chốt của primitive cứng I/O có thể dẫn đến hiện tượng chốt sai giá trị giữa các làn.
\end{enumerate}

\begin{figure}[H]
\centering
\IfFileExists{images/Timing.png}{%
    \includegraphics[width=0.85\textwidth]{images/Timing.png}%
}{%
    \fbox{\parbox[c][5cm][c]{0.8\textwidth}{\centering Không tìm thấy tệp images/Timing.png}}%
}
\caption{Mối quan hệ định thời giữa Clock và Data ở hai phương thức Edge-aligned (trên) và Center-aligned (dưới) trong giao tiếp truyền dữ liệu DDR ngõ ra (chuyển thể từ \cite{arumugham2000ddrtiming}).}

Nguyên tắc căn lề thời gian cần xem xét khi mở rộng giao tiếp DDR ra ngoài FPGA được minh họa chi tiết trên Hình~\ref{fig:center-aligned-timing} \cite{arumugham2000ddrtiming}. Trong giao tiếp đồng bộ nguồn (Source-Synchronous Output):
\begin{itemize}
    \item {Phương thức căn lề cạnh (Edge-aligned):} Cạnh của xung nhịp đồng hành (\path{ddr_clk_out}) chuyển trạng thái trùng khớp với thời điểm thay đổi của dữ liệu (\path{ddr_data}). Mô hình này ép phía nhận phải tự thực hiện dịch pha xung nhịp $90^\circ$ để có thể lấy mẫu dữ liệu an toàn.
    \item {Phương thức căn lề giữa (Center-aligned):} FPGA chủ động dịch pha xung nhịp ngõ ra $90^\circ$ (hoặc đảo pha $180^\circ$ qua khối ODDR) để cạnh lấy mẫu nằm giữa mắt dữ liệu (Data Eye). Cách này tối đa hóa biên an toàn định thời, bảo đảm $T_{suav}$ và $T_{hav}$ đều đáp ứng yêu cầu $t_{setup}$ ($T_{SU}$) và $t_{hold}$ ($T_{HD}$) của thiết bị thu.
\end{itemize}

Để Vivado thực hiện phân tích và tối ưu hóa các đường timing ngõ ra này một cách chính xác, tệp ràng buộc XDC cần khai báo đầy đủ clock tham chiếu và các tham số trễ ngoài chip. Trong Hình~\ref{lst:xdc-constraints}, xung nhịp truyền kèm \path{ddr_clk_out} được mô tả bằng lệnh \path{create_generated_clock} theo phương pháp ràng buộc của Vivado \cite{amd_ug903}.

Đối với DDR, các lệnh \path{set_output_delay} cần mô tả cả cạnh lên và cạnh xuống bằng các tùy chọn \path{-clock_fall} và \path{-add_delay}. Giá trị cực đại (\path{-max}) và cực tiểu (\path{-min}) được xác định từ yêu cầu thiết lập, giữ của thiết bị nhận và độ lệch trễ đường mạch in \cite{amd_ug903}. Các ràng buộc này cung cấp cơ sở để Vivado tính toán setup/hold slack cho toàn bộ giao tiếp ngõ ra.

\begin{lstlisting}[caption={Khai báo ràng buộc XDC hoàn chỉnh cho giao tiếp ngõ ra DDR (Source-Synchronous).}, label={lst:xdc-constraints}]
# 1. Clock he thong 100 MHz
create_clock -period 10.000 -name sys_clk [get_ports clk]

# 2. Forwarded output clock (ddr_clk_out)
create_generated_clock -name ddr_clk_out \
    -source [get_ports clk] -divide_by 1 [get_ports ddr_clk_out]

# 3. Output delay constraints for both DDR clock edges
set DDR_TSU 2.000; set DDR_THD 1.000
set DDR_SKEW_MAX 0.500; set DDR_SKEW_MIN 0.100

set OUT_MAX [expr {$DDR_TSU + $DDR_SKEW_MAX}]
set OUT_MIN [expr {-$DDR_THD - $DDR_SKEW_MIN}]
set DDR_BUS [get_ports {ddr_data[*] ddr_valid ddr_done}]

# Timing constraints for the rising edge
set_output_delay -clock ddr_clk_out -max $OUT_MAX $DDR_BUS
set_output_delay -clock ddr_clk_out -min $OUT_MIN $DDR_BUS

# Timing constraints for the falling edge
set_output_delay -clock ddr_clk_out -max $OUT_MAX $DDR_BUS -clock_fall -add_delay
set_output_delay -clock ddr_clk_out -min $OUT_MIN $DDR_BUS -clock_fall -add_delay
\end{lstlisting}
Tệp ràng buộc thời gian (XDC) cho khối giao tiếp DDR ngõ ra được xây dựng gồm 4 thành phần chính:
\begin{enumerate}
\item {Khai báo xung nhịp hệ thống và xung nhịp ngõ ra:} Lệnh \path{create_clock} thiết lập miền xung nhịp nội bộ \path{sys_clk} chạy ở tần số $100\text{ MHz}$ (chu kỳ $10\text{ ns}$). Lệnh \path{create_generated_clock} định nghĩa xung nhịp phát kèm \path{ddr_clk_out} ra chân chip. Việc tạo clock này giúp bộ phân tích Timing của Vivado tính toán chính xác độ lệch pha (Skew) giữa đường xung nhịp và đường dữ liệu khi truyền ra ngoài.
\item {Tính toán các thông số trễ ngõ ra (Output Delay):} Bốn biến TCL \path{DDR_TSU}, \path{DDR_THD}, \path{DDR_SKEW_MAX} và \path{DDR_SKEW_MIN} phản ánh các đặc tính phần cứng thực tế bên ngoài FPGA. Giá trị \path{OUT_MAX} ($2.5\text{ ns}$) biểu diễn tổng thời gian thiết lập yêu cầu tại bộ thu và độ lệch trễ đường dây lớn nhất. Giá trị \path{OUT_MIN} ($-1.1\text{ ns}$) thể hiện yêu cầu về thời gian giữ nhỏ nhất.
\item {Ràng buộc hai cạnh xung nhịp cho chuẩn DDR:} Hai lệnh \path{set_output_delay} đầu tiên áp dụng ràng buộc cho cạnh lên của \path{ddr_clk_out}. Hai lệnh tiếp theo sử dụng kết hợp hai cờ \path{-clock_fall} và \path{-add_delay} để mở rộng phân tích timing cho cạnh xuống. Điều này bắt buộc Vivado phải kiểm tra định thời (Setup/Hold slack) cho toàn bộ $2 \times N$ mẫu dữ liệu trong một chu kỳ clock.
\item {Mô phỏng mô hình tải vật lý I/O:} Các lệnh \path{set_load} ($5.0\text{ pF}$) và \path{set_max_transition} ($1.0\text{ ns}$) cung cấp mô hình tải thực tế của chân pin IOB, giúp công cụ tổng hợp ước lượng chính xác trễ tầng đệm ngõ ra (Pad Delay) và dạng sóng tín hiệu thực tế trên PCB.
\end{enumerate}

\section{Quy trình đóng gói và mô phỏng kiểm chứng mức RTL}

Để đảm bảo các khối ngoại vi nối tiếp (UART, SPI, I2C) và khối giao tiếp tốc độ cao DDR hoạt động chính xác theo đúng mô tả kiến trúc trước khi tiến hành tổng hợp mạch (Synthesis), quy trình kiểm chứng chức năng ở mức RTL (RTL Functional Verification) được thực hiện chặt chẽ thông qua môi trường Testbench tự động trên phần mềm Vivado Simulator (XSIM).

\subsection{Đóng gói các khối ngoại vi và tích hợp hệ thống (SoC Packaging)}

Quá trình đóng gói và tích hợp hệ thống SoC được thực hiện thông qua chiến lược thiết kế mô-đun hóa, đảm bảo tính liên kết chặt chẽ giữa lõi xử lý và các thiết bị ngoại vi. Khởi đầu, toàn bộ các khối ngoại vi nối tiếp được chuẩn hóa giao diện theo đặc tả AXI4-Lite (Slave Packaging), trong đó các nhóm tín hiệu được ánh xạ đầy đủ theo 5 kênh giao tiếp độc lập (AW, W, B, AR, R). Tiếp theo, các ngoại vi này được đấu nối trực tiếp vào bộ định tuyến AXI Crossbar trung tâm nằm trong mô-đun \path{soc_axi_top}. Bộ định tuyến này đảm nhiệm vai trò căn cứ vào dải địa chỉ trên kênh AW hoặc AR để chốt (Latch) và dẫn luồng giao dịch đến đúng thiết bị mục tiêu, qua đó triệt tiêu hoàn toàn nguy cơ chồng chéo tín hiệu. Đối với phân hệ gia tốc phần cứng, đầu ra dữ liệu song song từ khối xử lý được đưa qua bộ tuần tự hóa DDR và nối trực tiếp tới các ô nguyên thủy I/O cứng (\path{ODDR}) tại tầng đệm ngõ ra (\path{IOB}) nhằm tối ưu hóa độ trễ truyền dẫn.

Sau khi quy trình tổng hợp (Synthesis) hoàn tất, sơ đồ RTL của toàn bộ hệ thống SoC được trích xuất nhằm đánh giá trực quan các kết nối vật lý và kiến trúc đường truyền giữa các IP. Do quy mô của hệ thống tích hợp khá lớn, sơ đồ RTL mức tổng thể (Top-level RTL) được chia thành hai phần chính để thuận tiện cho việc phân tích và trình bày:

\subsection{Xây dựng môi trường mô phỏng Testbench}

Môi trường kiểm chứng (Testbench) được viết bằng Verilog HDL đóng vai trò là Master bus giả lập các chu kỳ ghi/đọc dữ liệu của vi xử lý RISC-V thông qua bus AXI4-Lite. Sơ đồ khối của môi trường kiểm chứng được thể hiện qua các thành phần chính:
\begin{itemize}
    \item {Khối tạo Xung nhịp và Reset (Clock/Reset Generator):} Tạo xung nhịp hệ thống \path{clk} tần số $100~\text{MHz}$ (chu kỳ $10~\text{ns}$) và tín hiệu Reset tích cực mức thấp \path{rst_n}.
    \item {Khối giả lập giao thức AXI Master (Task-based Driver):} Sử dụng các \path{task} trong Verilog (\path{axi_write} và \path{axi_read}) để khởi tạo các chu kỳ ghi/đọc thanh ghi theo đúng quy trình bắt tay VALID/READY của chuẩn AXI.
    \item {Mô hình thiết bị ngoại vi ngoài chip (Loopback/Slave Models):}
    \begin{itemize}
        \item \textit{UART:} Nối trực tiếp chân phát \path{uart_tx} sang chân nhận \path{uart_rx} (mô hình Loopback) để kiểm tra tính toàn vẹn của dữ liệu truyền/nhận bất đồng bộ.
        \item \textit{SPI:} Nối chân \path{spi_mosi} sang \path{spi_miso} hoặc ghép nối với mô hình Verilog của bộ nhớ SPI Flash ngoài.
        \item \textit{I2C:} Sử dụng điện trở kéo lên (Pull-up resistor) mô phỏng trên đường \path{sda} và \path{scl}, kết nối tới mô hình I2C EEPROM slave.
    \end{itemize}
\end{itemize}

\subsection{Kịch bản mô phỏng và Kết quả kiểm chứng (Simulation Scenarios)}

Quy trình mô phỏng được phân bổ thành 5 nhóm kịch bản kiểm thử độc lập nhằm xác thực toàn diện chức năng của hệ thống. Mỗi kịch bản được thiết kế với mục đích và phương pháp bơm dữ liệu cụ thể như sau:

\subsubsection{Kịch bản 1: Kiểm thử giao tiếp AXI UART}
\begin{itemize}
    \item \textbf{Mục đích:} Đánh giá khả năng phân giải tốc độ Baud, cấu trúc khung truyền bất đồng bộ và luồng dữ liệu hai chiều thông qua bus AXI.
    \item \textbf{Phương pháp thực hiện:}
        \begin{enumerate}
            \item \textit{TC1 (Baud Rate \& Framing Check):} Dùng \path{axi_write} ghi mẫu \path{0x55} vào thanh ghi phát \path{REG_TXDATA}. Theo dõi đường \path{uart_tx} để xác minh độ rộng bit và tính nguyên vẹn của Start/Stop bit.
            \item \textit{TC2 (Data Pattern TX):} Ghi mẫu ngẫu nhiên (ví dụ: \path{0xA5}), lấy mẫu ngõ ra tại giữa chu kỳ bit và so sánh với mẫu gốc để đánh giá khả năng chống nhiễu máy trạng thái.
            \item \textit{TC3 (AXI Status Register Verification):} Đọc liên tục \path{REG_STATUS} bằng \path{axi_read} trước, trong và sau quá trình truyền để kiểm tra cờ Ready/Busy (bit 0) được phần cứng lật trạng thái tự động.
            \item \textit{TC4 (UART Receive Data RX):} Giả lập một chuỗi khung truyền (ví dụ: \path{0x3C}) bơm trực tiếp vào chân \path{uart_rx}. Sau khi khung nhận hoàn tất, đọc \path{REG_RXDATA} để xác minh kết quả.
        \end{enumerate}
\end{itemize}

\subsubsection{Kịch bản 2: Kiểm thử giao tiếp AXI SPI}
\begin{itemize}
    \item \textbf{Mục đích:} Kiểm tra chế độ phân cực xung nhịp (CPOL) và khả năng truyền tải dữ liệu song công (Full-Duplex) tốc độ cao.
    \item \textbf{Phương pháp thực hiện:}
        \begin{enumerate}
            \item \textit{TC1 \& TC2 (CPOL \& Full-Duplex Data Transfer):} Thiết lập một biến nội bộ trong Testbench đóng vai trò là Slave giả lập (chứa mẫu \path{0x3C}). Master khởi tạo ghi \path{0xD6} vào \path{REG_SPI_TXDATA}. Kiểm tra \path{spi_sclk} giữ mức thấp khi rảnh (CPOL = 0) và xác minh dữ liệu đọc về từ \path{spi_miso} khớp đúng với \path{0x3C}.
            \item \textit{TC3 (Chip Select Register Operation):} Thực hiện ghi lần lượt giá trị 0 và 1 vào \path{REG_SPI_CS} thông qua lệnh \path{axi_write}. Đọc phản hồi bằng \path{axi_read} và quan sát sự thay đổi logic trên chân vật lý \path{spi_cs} tương ứng.
        \end{enumerate}
\end{itemize}

\subsubsection{Kịch bản 3: Kiểm thử giao tiếp AXI I2C}
\begin{itemize}
    \item \textbf{Mục đích:} Đánh giá khả năng tạo các điều kiện giao tiếp khắt khe (START, STOP) và kỹ thuật kéo/nhả bus vật lý thông qua thanh ghi AXI.
    \item \textbf{Phương pháp thực hiện:}
        \begin{enumerate}
            \item \textit{TC1 (AXI Data Register Read/Write):} Ghi/Đọc giá trị \path{0xA7} vào \path{REG_I2C_DATA} nhằm kiểm tra tính tương thích giữa thanh ghi nội bộ I2C và bus AXI.
            \item \textit{TC2 (START Condition):} Ghi bit khởi lệnh vào \path{REG_I2C_CMD}. Quan sát đường \path{sda_out} hạ mức thấp trong khi \path{scl_out} đang ở mức cao.
            \item \textit{TC3 (Write Byte \& ACK/NACK):} Testbench giả lập cực máng hở bằng cách can thiệp vào \path{sda_in} ngay tại chu kỳ thứ 9. Thực hiện hai lần lặp: lần 1 kéo \path{sda_in} = 0 (giả lập ACK), lần 2 thả nổi \path{sda_in} = 1 (giả lập NACK) và đọc \path{REG_I2C_STATUS} để đối chiếu cờ phản hồi.
            \item \textit{TC4 (STOP Condition):} Kích hoạt điều kiện kết thúc; xác minh \path{sda_out} nảy lên mức cao trong khi \path{scl_out} đang giữ mức cao.
        \end{enumerate}
\end{itemize}

\subsubsection{Kịch bản 4: Kiểm thử gia tốc xử lý song song DDR}
\begin{itemize}
    \item \textbf{Mục đích:} Xác thực luồng dữ liệu phân mảnh qua IDDR, tính toán song song, và đóng gói dữ liệu tại hai cạnh xung nhịp qua ODDR.
    \item \textbf{Phương pháp thực hiện:}
        \begin{enumerate}
            \item \textit{TC1 (AXI Trigger All Lanes):} Nạp đủ $4$ cặp toán hạng qua bus AXI, kích hoạt khởi chạy đồng bộ. Kiểm tra dữ liệu được dồn ra ngõ \path{ddr_data} đúng thứ tự ở cả cạnh lên và cạnh xuống.
            \item \textit{TC2 (Independent Lane Execution - IDDR Trigger):} Gửi từng cặp dữ liệu đan xen qua ngõ vào \path{ddr_data_in} (sử dụng cả cạnh lên và xuống). Kiểm chứng bộ định tuyến độc lập phân phối đúng luồng dữ liệu vào các Lane đang rảnh mà không bị kẹt bus.
            \item \textit{TC3 (Edge Isolation):} Truyền dữ liệu ngẫu nhiên ở sườn lên và chuỗi toàn $0$ ở sườn xuống. Xác nhận rằng mạch IDDR không bị rò rỉ dữ liệu giữa hai miền sườn xung.
            \item \textit{TC4 (Throughput Benchmark):} Xây dựng bộ đếm thời gian thực (bằng biến \texttt{\$realtime}). Đo thời gian hoàn thành 4 phép toán khi đi qua 1 Lane (chạy tuần tự bằng lệnh AXI Write) và so sánh với thời gian khi sử dụng 2 Lane DDR (kết hợp IDDR/ODDR). Tính toán hệ số Speedup Ratio nhằm chứng minh mức độ cải thiện thông lượng của phần cứng.
        \end{enumerate}
\end{itemize}

\subsubsection{Kịch bản 5: Kiểm thử mức tích hợp hệ thống (SoC Top-Level)}
\begin{itemize}
    \item \textbf{Mục đích:} Kiểm tra năng lực định tuyến của bộ AXI Crossbar và khả năng chịu tải khi tất cả các ngoại vi hoạt động đồng thời.
    \item \textbf{Phương pháp thực hiện:}
        \begin{enumerate}
            \item \textit{TC1 (AXI Decoder Access Check):} Truy xuất vào một dải địa chỉ ngoại vi ngẫu nhiên để xác nhận cơ chế giải mã (Decoder) hoạt động ổn định và chốt quyền truy cập đúng thiết bị.
            \item \textit{TC2 \& TC3 (Concurrent Hardware Output):} Khởi chạy liên tiếp các lệnh \path{axi_write} kích hoạt phát dữ liệu cho UART, SPI, và I2C trong cùng một thời điểm. Xây dựng bộ đếm số lượng chu kỳ xung nhịp mà 2 hoặc 3 ngõ ra vật lý (như \path{uart_tx}, \path{spi_sclk}, \path{scl_out}) cùng tích cực để chứng minh hệ thống đạt trạng thái xử lý đa nhiệm (Concurrent) thực sự.
            \item \textit{TC4 (Interleaved Cross-Peripheral Access):} Thực hiện chuỗi lệnh \path{axi_read} và \path{axi_write} thay đổi liên tục địa chỉ đích giữa UART, SPI, và I2C. Giám sát để đảm bảo máy trạng thái AXI Crossbar không bị treo (Deadlock) hay gửi nhầm tín hiệu trong lúc các ngoại vi vẫn đang xử lý nền.
        \end{enumerate}
\end{itemize}
